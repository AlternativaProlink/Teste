<script>
  var dados = <?!= dadosJson ?>;
  var grande = false;
  function alternar() {
    grande = !grande;
    var w = grande ? Math.max(1000, Math.min(screen.availWidth - 80, 1700)) : 1000;
    var h = grande ? Math.max(680, Math.min(screen.availHeight - 140, 1100)) : 680;
    google.script.host.setWidth(w); google.script.host.setHeight(h);
  }
  function esc(s) { return String(s == null ? '' : s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function k(l, v) { return '<div class="k"><span>' + l + '</span><b>' + esc(v || '—') + '</b></div>'; }
  (function () {
    var d = dados;
    document.getElementById('chips').innerHTML = '<span class="chip">' + esc(d.data) + '</span><span class="chip">' + esc(d.demanda) + '</span>';
    var h = '<div class="grid">' + k('Responsável', d.resp) + k('Local', d.local) + k('Transporte', d.transporte && d.transporte !== 'Não' ? d.transporte : 'Não') + '</div>';
    if (d.alerta) h += '<div class="alerta">' + esc(d.alerta) + '</div>';
    h += '<section><h3>Serviços</h3>';
    h += d.servicos.length ? d.servicos.map(function (s) { return '<div class="row"><span>' + esc(s.nome) + '</span><span class="q">' + esc(s.qtd) + '</span><span class="p">' + esc(s.pontos) + ' pts</span></div>'; }).join('') : '<p class="muted">Nenhum serviço.</p>';
    h += '</section>';
    if (d.ocorrencias.length) {
      h += '<section><h3>Imprevistos e fora da OS</h3>' + d.ocorrencias.map(function (o) {
        var s = (o.situacao || '').toLowerCase(), cls = s === 'resolvido' ? 'ok' : (s === 'parcial' ? 'parcial' : (s === 'pendente' ? 'pendente' : ''));
        var mapa = (o.lat && o.lng) ? ' · <a href="https://www.google.com/maps?q=' + o.lat + ',' + o.lng + '" target="_blank">ver no mapa</a>' : '';
        return '<div class="oc"><div class="t"><b>' + esc(o.tipo) + '</b><span>' + (o.quando ? '<span class="tag">' + esc(o.quando) + '</span> ' : '') + (o.situacao ? '<span class="tag ' + cls + '">' + esc(o.situacao) + '</span>' : '') + '</span></div>'
          + '<span class="muted">' + esc([o.local, o.descricao].filter(String).join(' · ')) + mapa + '</span></div>';
      }).join('') + '</section>';
    }
    if (d.obs) h += '<section><h3>Observações</h3><p class="muted" style="margin:0">' + esc(d.obs) + '</p></section>';
    document.getElementById('m').innerHTML = h;
    document.getElementById('f').innerHTML = '<span>Equipe <b style="font-size:13px">' + esc(d.equipe) + '</b></span><span>Pontos <b>' + esc(d.pontos) + '</b></span>';
  })();
</script>
