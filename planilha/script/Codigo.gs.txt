function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function norm_(s) {
  return String(s == null ? '' : s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/\s+/g, ' ').trim();
}
function ativo_(v) {
  var n = norm_(v);
  return !(n === 'nao' || n === 'n' || n === 'inativo' || n === 'inativa' || n === 'false' || n === '0');
}
function num_(v) {
  if (v === '' || v === null || v === undefined) return null;
  if (typeof v === 'number') return v;
  var s = String(v).replace(/[^\d,.\-]/g, '');
  if (s.indexOf(',') !== -1) s = s.replace(/\./g, '').replace(',', '.');
  var n = parseFloat(s);
  return isNaN(n) ? null : n;
}
function seguro_(v) {
  var s = String(v == null ? '' : v);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}
function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }
function aba_(nome) { return ss_().getSheetByName(nome); }
function lerBloco_(sheet, primeiraLinha, colIni, nCols) {
  var ultima = sheet.getLastRow();
  if (ultima < primeiraLinha) return [];
  return sheet.getRange(primeiraLinha, colIni, ultima - primeiraLinha + 1, nCols).getValues();
}
function chaveDia_(d) { return d.getFullYear() + '-' + (d.getMonth() + 1) + '-' + d.getDate(); }
function soDia_(d) { return new Date(d.getFullYear(), d.getMonth(), d.getDate()); }
function fmtNum_(n) {
  var r = Math.round(n * 10) / 10;
  var s = (Math.abs(r % 1) > 0 ? r.toFixed(1) : String(Math.round(r))).replace('.', ',');
  return s.replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}
function fmtR_(n) { return 'R$ ' + Number(n || 0).toFixed(2).replace('.', ',').replace(/\B(?=(\d{3})+(?!\d))/g, '.'); }
function fmtData_(d) { return Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd/MM/yyyy'); }
function ts_(o) { return SpreadsheetApp.newTextStyle()
  .setFontFamily(FONTE).setFontSize(o.size || 10).setBold(!!o.bold).setForegroundColor(o.color || COR.texto).build(); }

function nr_(nome) { var r = ss_().getRangeByName(nome); return r; }
function nrVal_(nome, def) {
  var r = nr_(nome); if (!r) return def;
  var v = r.getValue(); return (v === '' || v === null) ? def : v;
}
function lerConfig_() {
  var c = {
    nome: String(nrVal_('CONF_NOME', 'Pro Link Serviços')),
    setor: String(nrVal_('CONF_SETOR', 'Infraestrutura')),
    pin: String(nrVal_('CONF_PIN', '')).trim(),
    anotar: ativo_(nrVal_('CONF_ANOTAR', 'Sim')),
    transporte: ativo_(nrVal_('CONF_TRANSP', 'Sim')),
    diasRetro: num_(nrVal_('CONF_RETRO', 7)) || 7,
    diasOS: num_(nrVal_('CONF_OSREP', 30)) || 30,
    periodo: String(nrVal_('CONF_PERIODO', 'Mês atual')),
    equipePadrao: String(nrVal_('CONF_EQPADRAO', 'Todas as equipes')),
    quadros: [], medidas: [], ajustes: {}
  };
  var q = nr_('CONF_QUADROS');
  if (q) q.getValues().forEach(function (r) { if (r[1]) c.quadros.push({ ver: r[0] === true, nome: String(r[1]), ordem: num_(r[2]) || 99 }); });
  if (!c.quadros.length) c.quadros = QUADROS_PADRAO_.map(function (n, i) { return { ver: true, nome: n, ordem: i + 1 }; });
  var m = nr_('CONF_MEDIDAS');
  if (m) m.getValues().forEach(function (r) {
    if (!r[0]) return;
    c.medidas.push({ chaves: String(r[0]).split('·').map(norm_).filter(String), regra: String(r[5] || ''), mpp: num_(r[6]), bloco: num_(r[7]) });
  });
  var a = nr_('CONF_AJUSTES');
  if (a) a.getValues().forEach(function (r) { if (r[0]) c.ajustes[norm_(r[0])] = (num_(r[5]) === null ? 100 : num_(r[5])); });
  return c;
}
var QUADROS_PADRAO_ = ['Indicadores', 'Serviços mais executados', 'Ocorrências mais frequentes', 'Metas do mês por equipe', 'Pendências para retorno', 'Comissão por técnico (com equipe filtrada)'];

function lerLista_(sheet, def) {
  var colIni = 999, colFim = 0, k;
  for (k in def) { colIni = Math.min(colIni, def[k]); colFim = Math.max(colFim, def[k]); }
  var linhas = lerBloco_(sheet, CFG.LINHA_CAD, colIni, colFim - colIni + 1);
  var out = [];
  for (var i = 0; i < linhas.length; i++) {
    var nome = String(linhas[i][def.nome - colIni] || '').trim();
    if (!nome) continue;
    var item = { nome: nome };
    for (k in def) if (k !== 'nome') item[k] = linhas[i][def[k] - colIni];
    item.ativoFlag = ativo_(item.ativo);
    out.push(item);
  }
  return out;
}
function lerCadastros_() {
  var sh = aba_(CFG.ABA_CAD);
  if (!sh) throw new Error('Aba ' + CFG.ABA_CAD + ' não encontrada');
  return { servicos: lerLista_(sh, CFG.SERV), ocorrencias: lerLista_(sh, CFG.OCC), transportes: lerLista_(sh, CFG.TRANSP), demandas: lerLista_(sh, CFG.DEM) };
}
function lerEquipes_() {
  var sh = aba_(CFG.ABA_EQUIPES);
  if (!sh) throw new Error('Aba ' + CFG.ABA_EQUIPES + ' não encontrada');
  var linhas = sh.getRange(CFG.LINHA_EQUIPES, CFG.EQ.nome, CFG.EQ_MAX_LINHAS, 5).getValues();
  var out = [];
  linhas.forEach(function (r) {
    var nome = String(r[0] || '').trim();
    if (!nome) return;
    var membros = [String(r[2] || '').trim(), String(r[3] || '').trim()].filter(String);
    out.push({ nome: nome, meta: num_(r[1]), membros: membros, ativa: ativo_(r[4]) });
  });
  return out;
}
function lerEscala_() {
  var sh = aba_(CFG.ABA_EQUIPES); if (!sh) return [];
  var v = sh.getRange(CFG.ESCALA.linha, 7, CFG.ESCALA.qtd, 5).getValues();
  var out = [];
  v.forEach(function (r) {
    var tec = String(r[0] || '').trim(), tipo = norm_(r[1]);
    if (!tec || !tipo || !(r[3] instanceof Date)) return;
    var ate = (r[4] instanceof Date) ? r[4] : r[3];
    out.push({ tecnico: tec, tipo: tipo.indexOf('refor') === 0 ? 'reforco' : (tipo.indexOf('ferias') === 0 ? 'ferias' : 'folga'),
      equipe: String(r[2] || '').trim(), de: soDia_(r[3]), ate: soDia_(ate) });
  });
  return out;
}
function lerRegrasComissao_() {
  var teto = num_(nrVal_('EQ_TETO', 200)) || 200;
  var faixas = [];
  var r = nr_('EQ_FAIXAS');
  if (r) r.getValues().forEach(function (x) { var de = num_(x[0]), v = num_(x[2]); if (de !== null && v !== null) faixas.push({ de: de, ate: num_(x[1]), valor: v }); });
  if (!faixas.length) faixas = [{ de: 800, ate: 899, valor: 75 }, { de: 900, ate: 999, valor: 100 }, { de: 1000, ate: 1199, valor: 150 }, { de: 1200, ate: null, valor: 200 }];
  faixas.sort(function (a, b) { return a.de - b.de; });
  return { teto: teto, faixas: faixas };
}
// quem trabalha em cada equipe numa data, considerando a escala
function equipesNaData_(equipes, escala, data) {
  var d = soDia_(data).getTime(), mapa = {};
  var fora = {}, reforcos = [];
  escala.forEach(function (e) {
    if (d < e.de.getTime() || d > e.ate.getTime()) return;
    fora[norm_(e.tecnico)] = true;
    if (e.tipo === 'reforco' && e.equipe) reforcos.push(e);
  });
  equipes.forEach(function (t) {
    mapa[t.nome] = t.membros.filter(function (m) { return !fora[norm_(m)]; }).map(function (m) { return { nome: m, reforco: false }; });
  });
  reforcos.forEach(function (e) { if (mapa[e.equipe]) mapa[e.equipe].push({ nome: e.tecnico, reforco: true }); });
  return mapa;
}

function getPin_() {
  var p = nr_('CONF_PIN');
  if (p) return String(p.getValue() == null ? '' : p.getValue()).trim();
  return '';
}
function pinValido_(pinRecebido) {
  var pin = getPin_();
  return !pin || String(pinRecebido == null ? '' : pinRecebido).trim() === pin;
}

/* ============================================================
 *  doGet — listas para o app
 * ============================================================ */
function doGet(e) {
  try {
    var p = (e && e.parameter) || {};
    if (!pinValido_(p.pin)) return json_({ ok: false, status: 'error', code: 'pin', message: 'PIN inválido ou não informado.' });
    var cad = lerCadastros_(), eq = lerEquipes_(), conf = lerConfig_();
    var hoje = equipesNaData_(eq, lerEscala_(), new Date());
    var equipes = {}, ordem = [];
    eq.forEach(function (t) { if (t.ativa) { equipes[t.nome] = (hoje[t.nome] || []).map(function (m) { return m.nome; }); ordem.push(t.nome); } });
    return json_({
      ok: true, status: 'success', geradoEm: new Date().toISOString(), pinObrigatorio: !!getPin_(),
      ordemEquipes: ordem, equipes: equipes,
      demandas: cad.demandas.filter(function (d) { return d.ativoFlag; }).map(function (d) { return d.nome; }),
      servicos: cad.servicos.filter(function (s) { return s.ativoFlag; }).map(function (s) {
        return { n: s.nome, c: String(s.categoria || '').trim() || 'Outros', u: String(s.medida || '').trim() || 'unidade' };
      }),
      ocorrencias: cad.ocorrencias.filter(function (o) { return o.ativoFlag; }).map(function (o) { return o.nome; }),
      transportes: cad.transportes.filter(function (t) { return t.ativoFlag; }).map(function (t) { return t.nome; }),
      app: { anotarNoCaminho: conf.anotar, mostrarTransporte: conf.transporte, diasRetroativo: conf.diasRetro, diasOSRepetida: conf.diasOS }
    });
  } catch (err) {
    return json_({ ok: false, status: 'error', code: 'server', message: String(err) });
  }
}

/* ============================================================
 *  doPost — recebe um registro
 * ============================================================ */
function doPost(e) {
  var lock = LockService.getScriptLock(), temLock = false, gravou = false, resposta;
  try {
    var data = JSON.parse(e.postData.contents);
    if (!pinValido_(data.pin)) return json_({ ok: false, status: 'error', code: 'pin', message: 'PIN inválido ou não informado.' });

    var ordemServico = data.os || data.nOS || data.numero_os || data.ordemServico || '';
    var equipe = data.equipe || '', responsavel = data.responsavel || '';
    var local = data.local || data.cidade || '', demanda = data.demanda || '';
    var servicos = data.servicos || '', ocorrencias = data.ocorrencias || '';
    var transporte = data.transporte || '', observacoes = data.observacoes || '';
    var detalhe = data.ocorrenciasDetalhe || '';
    var idRegistro = String(data.id || '').trim();

    if (!String(ordemServico).trim() || !equipe || !responsavel || (!servicos && (!ocorrencias || ocorrencias === 'Nenhuma'))) {
      return json_({ ok: false, status: 'error', code: 'invalid', message: 'Registro incompleto (OS, equipe, responsável e serviços são obrigatórios).' });
    }
    var dataFormatada = new Date();
    if (data.data) {
      var partes = String(data.data).split('T')[0].split('-');
      if (partes.length === 3) dataFormatada = new Date(+partes[0], +partes[1] - 1, +partes[2]);
    }
    try { lock.waitLock(25000); temLock = true; }
    catch (le) { return json_({ ok: false, status: 'error', code: 'busy', message: 'Servidor ocupado. Tente novamente.' }); }

    var sheet = aba_(CFG.ABA_REGISTROS);
    if (!sheet) return json_({ ok: false, status: 'error', code: 'server', message: 'Aba Registros não encontrada' });

    var ultima = Math.max(sheet.getLastRow(), CFG.LINHA_REG);
    var qtd = ultima - CFG.LINHA_REG + 1;
    var colH = sheet.getRange(CFG.LINHA_REG, 8, qtd, 1).getValues();
    var colS = sheet.getRange(CFG.LINHA_REG, CFG.REG_COL_ID, qtd, 1).getValues();
    if (idRegistro) {
      for (var j = 0; j < colS.length; j++) {
        if (String(colS[j][0]).trim() === idRegistro) return json_({ ok: true, status: 'success', duplicate: true, linha: CFG.LINHA_REG + j, message: 'Registro já estava salvo.' });
      }
    }
    var ultimaReal = CFG.LINHA_REG - 1;
    for (var i = colH.length - 1; i >= 0; i--) { if (colH[i][0] !== '' && colH[i][0] !== null) { ultimaReal = CFG.LINHA_REG + i; break; } }
    var linha = ultimaReal + 1;
    if (linha > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), 50);

    var calc = { pontos: 0, alertas: [] };
    try { calc = calcularPontos_(servicos, ocorrencias, transporte); }
    catch (ce) { calc.alertas.push('Erro ao calcular pontos: ' + ce); }

    sheet.getRange(linha, 7, 1, CFG.REG_COLS).setValues([[
      dataFormatada, seguro_(ordemServico), equipe, responsavel, seguro_(local), demanda,
      seguro_(servicos), seguro_(ocorrencias), seguro_(transporte), seguro_(observacoes),
      calc.pontos, '', idRegistro, resumoAlertas_(calc.alertas), seguro_(detalhe)
    ]]);
    sheet.getRange(linha, CFG.REG_COL_ALERTA).setNote(calc.alertas.join('\n'));
    sheet.getRange(linha, 7).setNumberFormat('dd/MM/yyyy');
    SpreadsheetApp.flush();
    gravou = true;
    resposta = { ok: true, status: 'success', message: 'Salvo com sucesso!', linha: linha, pontos: calc.pontos, alerta: calc.alertas.join(' | ') };
  } catch (error) {
    return json_({ ok: false, status: 'error', code: 'server', message: String(error) });
  } finally {
    if (temLock) lock.releaseLock();
  }
  if (gravou) {
    try { atualizarPainel(); } catch (e1) { console.error('Painel: ' + e1); }
    try { atualizarRelatorio(); } catch (e2) { console.error('Relatório: ' + e2); }
  }
  return json_(resposta);
}

// "Tipo [Quando · Situação] @ local: descrição → Serviços: ..."  ou  "Tipo: descrição"
function tipoOcorrencia_(txt) {
  var t = String(txt || '').trim();
  var cortes = [t.indexOf(' ['), t.indexOf(' @ '), t.indexOf(':')].filter(function (i) { return i > 0; });
  return (cortes.length ? t.slice(0, Math.min.apply(null, cortes)) : t).trim();
}
function situacaoOcorrencia_(txt) {
  var m = String(txt || '').match(/\[([^\]]*)\]/);
  if (!m) return '';
  var p = m[1].split('·').map(function (x) { return x.trim(); });
  return p[1] || '';
}
function quandoOcorrencia_(txt) {
  var m = String(txt || '').match(/\[([^\]]*)\]/);
  return m ? (m[1].split('·')[0] || '').trim() : '';
}
function partesServico_(txt) {
  var s = String(txt || '').trim(), nome = s, qtd = 1, un = '';
  var m = s.match(/^(.*)\s\(\s*([\d.,]+)\s*([^()]*)\)$/);
  if (m) { nome = m[1].trim(); qtd = num_(m[2]) || 0; un = String(m[3] || '').trim(); }
  return { nome: nome, qtd: qtd, un: un };
}
function regraMedida_(conf, medida) {
  var n = norm_(medida || 'unidade');
  for (var i = 0; i < conf.medidas.length; i++) {
    if (conf.medidas[i].chaves.indexOf(n) !== -1) return conf.medidas[i];
  }
  if (n === 'm' || n.indexOf('metro') === 0) {
    for (var j = 0; j < conf.medidas.length; j++) if (conf.medidas[j].chaves.indexOf('metro') !== -1) return conf.medidas[j];
    return { regra: 'Por metro', mpp: 40, bloco: 100 };
  }
  return null;
}
function pontosMetro_(regra, metros) {
  var mpp = regra.mpp || 40, b = regra.bloco || (norm_(regra.regra).indexOf('os') !== -1 ? 500 : 100);
  var r = norm_(regra.regra);
  if (r.indexOf('bloco') !== -1) return Math.floor(metros / b) * (b / mpp);
  if (r.indexOf('por os') !== -1) return Math.ceil(metros / b) * (b / mpp);
  return metros / mpp;
}
function pontosItem_(item, cadItem, conf, alertas, rotulo) {
  var medida = String((cadItem && cadItem.medida) || item.un || 'unidade');
  var regra = regraMedida_(conf, medida);
  var aj = conf.ajustes[norm_(cadItem ? cadItem.nome : item.nome)];
  var fator = (aj === undefined ? 100 : aj) / 100;
  var nm = norm_(medida);
  if (nm === 'm' || nm.indexOf('metro') === 0) {
    return pontosMetro_(regra || { regra: 'Por metro', mpp: 40 }, item.qtd) * fator;
  }
  var pts = cadItem ? num_(cadItem.pontos) : null;
  if (pts === null) { alertas.push(rotulo + ' sem pontos: ' + (cadItem ? cadItem.nome : item.nome)); return 0; }
  if (regra && norm_(regra.regra).indexOf('fixo') !== -1) return pts * fator;
  return pts * item.qtd * fator;
}
function calcularPontos_(servicosTexto, ocorrenciasTexto, transporteTexto, cadCache, confCache) {
  var cad = cadCache || lerCadastros_(), conf = confCache || lerConfig_();
  var total = { pontos: 0, alertas: [] };
  function mapa(lista) { var m = {}; lista.forEach(function (it) { m[norm_(it.nome)] = it; }); return m; }
  var mServ = mapa(cad.servicos), mOcc = mapa(cad.ocorrencias), mTr = mapa(cad.transportes);

  if (servicosTexto) String(servicosTexto).split('|').forEach(function (parte) {
    if (!parte.trim()) return;
    var it = partesServico_(parte), c = mServ[norm_(it.nome)];
    if (!c) { total.alertas.push('Serviço sem cadastro: ' + it.nome); return; }
    total.pontos += pontosItem_(it, c, conf, total.alertas, 'Serviço');
  });
  if (ocorrenciasTexto && ocorrenciasTexto !== 'Nenhuma') String(ocorrenciasTexto).split('|').forEach(function (parte) {
    if (!parte.trim()) return;
    var nome = tipoOcorrencia_(parte), c = mOcc[norm_(nome)];
    if (!c) { total.alertas.push('Ocorrência sem cadastro: ' + nome); return; }
    total.pontos += pontosItem_({ nome: nome, qtd: 1 }, c, conf, total.alertas, 'Ocorrência');
  });
  if (transporteTexto && transporteTexto !== 'Não' && transporteTexto !== 'Nenhum') {
    var tx = String(transporteTexto).trim(), nomeT = tx, viagens = 1;
    var mt = tx.match(/^(.*)\s\(\s*(\d+)\s*viage[mn]/i);
    if (mt) { nomeT = mt[1].trim(); viagens = parseInt(mt[2], 10) || 1; }
    var cT = mTr[norm_(nomeT)];
    if (!cT) total.alertas.push('Transporte sem cadastro: ' + nomeT);
    else total.pontos += pontosItem_({ nome: nomeT, qtd: viagens }, cT, conf, total.alertas, 'Transporte');
  }
  total.pontos = Math.round(total.pontos * 100) / 100;
  return total;
}
function resumoAlertas_(alertas) {
  if (!alertas || !alertas.length) return '';
  var semCad = alertas.filter(function (a) { return a.indexOf('sem cadastro') !== -1; }).length;
  var semPts = alertas.filter(function (a) { return a.indexOf('sem pontos') !== -1; }).length;
  var outros = alertas.length - semCad - semPts, p = [];
  if (semPts) p.push(semPts + (semPts > 1 ? ' itens sem pontos' : ' item sem pontos'));
  if (semCad) p.push(semCad + ' sem cadastro');
  if (outros) p.push(outros + (outros > 1 ? ' avisos' : ' aviso'));
  return '⚠ ' + p.join(' · ');
}

function lerRegistros_() {
  var sh = aba_(CFG.ABA_REGISTROS);
  var v = lerBloco_(sh, CFG.LINHA_REG, 7, CFG.REG_COLS);
  var out = [];
  v.forEach(function (r, i) {
    var d = r[0] instanceof Date ? r[0] : (r[0] ? new Date(r[0]) : null);
    if (!d || isNaN(d.getTime()) || !r[1] || !String(r[2]).trim()) return;
    var det = null;
    if (r[14]) { try { det = JSON.parse(String(r[14]).replace(/^'/, '')); } catch (e) { det = null; } }

function ocorrenciasDoRegistro_(reg) {
  if (reg.detalhe && reg.detalhe.length) return reg.detalhe.map(function (o) {
    return { tipo: o.tipo, quando: o.quando || '', situacao: o.situacao || '', local: o.local || '', descricao: o.descricao || '',
      lat: o.lat, lng: o.lng, servicos: o.servicos || [] };
  });
  var t = String(reg.ocorrencias || '');
  if (!t || t === 'Nenhuma') return [];
  return t.split('|').map(function (p) {
    p = p.trim();
    var loc = (p.match(/ @ ([^:]*):/) || [])[1] || '';
    var desc = p.indexOf(':') !== -1 ? p.slice(p.indexOf(':') + 1).split('→')[0].trim() : '';
    return { tipo: tipoOcorrencia_(p), quando: quandoOcorrencia_(p), situacao: situacaoOcorrencia_(p), local: loc.trim(), descricao: desc };
  }).filter(function (o) { return o.tipo; });
}
function contarServicos_(servStr, mapa) {
  var soma = 0;
  String(servStr || '').split('|').forEach(function (s) {
    s = s.trim(); if (!s) return;
    var it = partesServico_(s), un = norm_(it.un);
    var metragem = un ? (un === 'm' || un.indexOf('metro') === 0) : /cabo|lancamento|trecho|cordoalha/.test(norm_(it.nome));
    var q = metragem ? 1 : (it.qtd || 1);
    soma += q;
    if (mapa) { var k = norm_(it.nome); if (!mapa[k]) mapa[k] = { desc: it.nome, qtd: 0 }; mapa[k].qtd += q; }
  });
  return soma;
}

/* ============================================================
 *  Comissão por técnico (faixas, teto, escala)
 * ============================================================ */
function calcComissoes_(ano, mes, regsCache) {
  var equipes = lerEquipes_(), escala = lerEscala_(), regras = lerRegrasComissao_();
  var regs = (regsCache || lerRegistros_()).filter(function (r) { return r.data.getFullYear() === ano && r.data.getMonth() + 1 === mes; });
  var porEqDia = {};
  regs.forEach(function (r) { var k = r.equipe + '|' + chaveDia_(r.data); if (!porEqDia[k]) porEqDia[k] = { equipe: r.equipe, data: r.data, pts: 0 }; porEqDia[k].pts += r.pontos; });
  var pessoas = {};
  function p_(nome, base) { var k = norm_(nome); if (!pessoas[k]) pessoas[k] = { nome: nome, equipeBase: base || '', dias: {}, diasReforco: {}, equipesReforco: {}, pontos: 0, ausencias: [] }; return pessoas[k]; }
  equipes.forEach(function (t) { t.membros.forEach(function (m) { p_(m, t.nome); }); });
  Object.keys(porEqDia).forEach(function (k) {
    var x = porEqDia[k], mapa = equipesNaData_(equipes, escala, x.data), pres = mapa[x.equipe] || [];
    if (!pres.length) return;
    var parte = x.pts * 2 / pres.length;
    pres.forEach(function (m) {
      var pe = p_(m.nome, ''); pe.pontos += parte; pe.dias[chaveDia_(x.data)] = true;
      if (m.reforco) { pe.diasReforco[chaveDia_(x.data)] = true; pe.equipesReforco[x.equipe] = true; }
    });
  });
  var ini = new Date(ano, mes - 1, 1).getTime(), fim = new Date(ano, mes, 0).getTime();
  escala.forEach(function (e) {
    if (e.ate.getTime() < ini || e.de.getTime() > fim) return;
    var a = Math.max(e.de.getTime(), ini), b = Math.min(e.ate.getTime(), fim), dias = Math.round((b - a) / 86400000) + 1;
    var pe = p_(e.tecnico, '');
    if (e.tipo !== 'reforco') pe.ausencias.push((e.tipo === 'ferias' ? 'Férias ' : 'Folga ') + dias + (dias > 1 ? ' dias' : ' dia'));
  });
  return Object.keys(pessoas).map(function (k) {
    var pe = pessoas[k], pts = Math.round(pe.pontos * 10) / 10, faixa = null;
    regras.faixas.forEach(function (f) { if (pts >= f.de) faixa = f; });
    var valor = faixa ? Math.min(faixa.valor, regras.teto) : 0, prox = null;
    regras.faixas.forEach(function (f) { if (!prox && f.de > pts) prox = f; });
    var obs = pe.ausencias.slice();
    var nReforco = Object.keys(pe.diasReforco).length;
    if (nReforco) obs.push('Reforço na ' + Object.keys(pe.equipesReforco).join(', ') + ' (' + nReforco + (nReforco > 1 ? ' dias)' : ' dia)'));
    return { nome: pe.nome, equipeBase: pe.equipeBase, equipesReforco: Object.keys(pe.equipesReforco), dias: Object.keys(pe.dias).length,
      pontos: pts, faixa: faixa ? (faixa.de + (faixa.ate ? '–' + faixa.ate : '+')) : 'abaixo de ' + regras.faixas[0].de,
      valor: valor, faltam: prox ? Math.ceil(prox.de - pts) : 0, obs: obs.join(' · '), teto: regras.teto };
  }).sort(function (a, b) { return b.pontos - a.pontos; });
}

function recalcularPontos() {
  var ui = SpreadsheetApp.getUi();
  var r = ui.prompt('Recalcular pontos', 'Informe o mês/ano (ex.: 11/2026) ou deixe em branco para recalcular todos os registros.', ui.ButtonSet.OK_CANCEL);
  if (r.getSelectedButton() !== ui.Button.OK) return;
  var t = r.getResponseText().trim(), mes = null, ano = null;
  if (t) { var m = t.match(/^(\d{1,2})\s*\/\s*(\d{4})$/); if (!m) { ui.alert('Formato inválido. Use mm/aaaa.'); return; } mes = +m[1]; ano = +m[2]; }
  var n = recalcular_(ano, mes);
  try { atualizarPainel(); atualizarRelatorio(); } catch (e) {}
  ui.alert('Pontos recalculados em ' + n + (n === 1 ? ' registro.' : ' registros.'));
}
function recalcular_(ano, mes) {
  var sh = aba_(CFG.ABA_REGISTROS), last = sh.getLastRow();
  if (last < CFG.LINHA_REG) return 0;
  var v = sh.getRange(CFG.LINHA_REG, 7, last - CFG.LINHA_REG + 1, CFG.REG_COLS).getValues();
  var cad = lerCadastros_(), conf = lerConfig_(), pts = [], com = [], al = [], notas = [], n = 0;
  v.forEach(function (r) {
    var d = r[0] instanceof Date ? r[0] : null, alvo = d && r[1] && (!ano || (d.getFullYear() === ano && d.getMonth() + 1 === mes));
    if (alvo) {
      var c = calcularPontos_(r[6], r[7], r[8], cad, conf);
      pts.push([c.pontos]); com.push(['']); al.push([resumoAlertas_(c.alertas)]); notas.push([c.alertas.join('\n')]); n++;
    } else { pts.push([r[10]]); com.push([r[11]]); al.push([r[13]]); notas.push([null]); }
  });
  sh.getRange(CFG.LINHA_REG, CFG.REG_COL_PONTOS, pts.length, 1).setValues(pts);
  sh.getRange(CFG.LINHA_REG, CFG.REG_COL_COMISSAO, com.length, 1).setValues(com);
  var rAl = sh.getRange(CFG.LINHA_REG, CFG.REG_COL_ALERTA, al.length, 1);
  var notasAtuais = rAl.getNotes();
  rAl.setValues(al);
  rAl.setNotes(notas.map(function (x, i) { return [x[0] === null ? notasAtuais[i][0] : x[0]]; }));
  return n;
}

function lerFiltroPainel_(sh) {
  var mesTxt = String(sh.getRange('Q5').getValue()), ano = Number(sh.getRange('T5').getValue()) || new Date().getFullYear();
  var mes = MESES.indexOf(mesTxt) + 1; // 0 = todos
  var eq = String(sh.getRange('W5').getValue() || 'Todas as equipes');

function atualizarPainel() {
  var sh = aba_(CFG.ABA_PAINEL); if (!sh) return;
  var f = lerFiltroPainel_(sh), conf = lerConfig_(), regsTodos = lerRegistros_(), equipes = lerEquipes_();
  var regsPeriodo = regsTodos.filter(function (r) { return r.data.getFullYear() === f.ano && (!f.mes || r.data.getMonth() + 1 === f.mes); });
  var regs = regsPeriodo.filter(function (r) { return !f.equipe || r.equipe === f.equipe; });

  var servMap = {}, ocoMap = {}, dias = {}, totServ = 0, totOco = 0, transp = 0, desvio = 0, pts = 0, pend = [];
  regs.forEach(function (r) {
    totServ += contarServicos_(r.servicos, servMap);
    dias[chaveDia_(r.data)] = true;
    pts += r.pontos;
    ocorrenciasDoRegistro_(r).forEach(function (o) {
      totOco++;
      var k = norm_(o.tipo); if (!ocoMap[k]) ocoMap[k] = { desc: o.tipo, qtd: 0 }; ocoMap[k].qtd++;
      var s = norm_(o.situacao);
      if (s === 'pendente' || s === 'parcial') pend.push({ os: r.os, equipe: r.equipe, tipo: o.tipo, local: o.local, sit: o.situacao, lat: o.lat, lng: o.lng, data: r.data });
    });
    var t = String(r.transporte || '');
    if (t && t !== 'Não' && t !== 'Nenhum') { transp++; var mv = t.match(/\(\s*(\d+)\s*viage/i); if (mv && +mv[1] > 1) desvio++; }
  });
  var nDias = Object.keys(dias).length;
  var vis = {}; conf.quadros.forEach(function (q) { vis[q.nome] = q; });
  var kpiVisivel = !vis['Indicadores'] || vis['Indicadores'].ver;

  // indicadores
  var kpis = [
    ['Total de serviços', totServ, 'em ' + regs.length + ' OS'],
    ['Média diária', nDias ? Math.round(totServ / nDias) : 0, nDias + (nDias === 1 ? ' dia com registro' : ' dias com registro')],
    ['Ocorrências', totOco, 'no período'],
    ['Transportes de material', transp, desvio + (desvio === 1 ? ' com desvio de rota' : ' com desvio de rota')],
    ['Pontos no período', fmtNum_(pts), f.equipe || 'todas as equipes'],
    ['Pendências', pend.length, 'para retorno']
  ];
  PAINEL.kpiCols.forEach(function (c, i) {
    sh.getRange(7, c).setValue('         ' + kpis[i][0]);
    sh.getRange(8, c).setValue(kpis[i][1]);
    sh.getRange(9, c).setValue(kpis[i][2]);
  });
  if (kpiVisivel) sh.showRows(6, 5); else sh.hideRows(6, 5);

  // quadros nos espaços
  var lista = conf.quadros.filter(function (q) { return q.ver && q.nome !== 'Indicadores'; }).sort(function (a, b) { return a.ordem - b.ordem; }).map(function (q) { return q.nome; });
  var temComissao = lista.indexOf(QUADROS_PADRAO_[5]) !== -1;
  if (f.equipe && temComissao) { var im = lista.indexOf('Metas do mês por equipe'); if (im !== -1) lista[im] = QUADROS_PADRAO_[5]; lista = lista.filter(function (n, i) { return lista.indexOf(n) === i; }); }
  else lista = lista.filter(function (n) { return n !== QUADROS_PADRAO_[5]; });

  var dados = {
    'Serviços mais executados': function (slot) { ranking_(sh, slot, 'Serviços mais executados', servMap, COR.verde, 'Nenhum serviço no período.'); },
    'Ocorrências mais frequentes': function (slot) { ranking_(sh, slot, 'Ocorrências mais frequentes', ocoMap, '#c26a2b', 'Nenhuma ocorrência no período.'); },
    'Metas do mês por equipe': function (slot) { quadroMetas_(sh, slot, equipes, regsPeriodo, f); },
    'Pendências para retorno': function (slot) { quadroPendencias_(sh, slot, pend); },
  };
  dados[QUADROS_PADRAO_[5]] = function (slot) { quadroComissao_(sh, slot, f, regsTodos); };
  PAINEL.slots.forEach(function (slot, i) {
    limparSlot_(sh, slot);
    var nome = lista[i];
    if (nome && dados[nome]) dados[nome](slot);
  });
  [[0, 1, 11], [2, 3, 23]].forEach(function (b) {
    if (lista[b[0]] || lista[b[1]]) sh.showRows(b[2], 11); else sh.hideRows(b[2], 11);
  });
  sh.getRange('Q2').setValue(conf.setor + ' · atualizado às ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm'));
}
function limparSlot_(sh, s) {
  var r = sh.getRange(s.r, s.c, 11, 8);
  r.breakApart(); r.clearContent(); r.clearNote(); r.setBorder(false, false, false, false, false, false);
  r.setBackground(COR.fundo).setFontWeight('normal').setFontSize(10).setFontColor(COR.texto).setHorizontalAlignment('left');
}
function moldeSlot_(sh, s, titulo, sub) {
  card_(sh, sh.getRange(s.r, s.c, 11, 8).getA1Notation());
  sh.getRange(s.r, s.c, 1, 5).merge().setValue(titulo).setFontSize(11).setFontWeight('bold').setFontColor(COR.verdeEsc).setVerticalAlignment('middle');
  if (sub) sh.getRange(s.r, s.c + 5, 1, 3).merge().setValue(sub).setFontSize(8).setFontColor('#6b7a6f').setHorizontalAlignment('right');
}
function linhaSep_(sh, s, row) { sh.getRange(row, s.c, 1, 8).setBorder(null, null, true, null, null, null, COR.linha, SpreadsheetApp.BorderStyle.SOLID); }
function barra_(valor, max, cor) {
  return '=SPARKLINE(' + Math.max(0, valor) + ',{"charttype","bar";"max",' + Math.max(max, 1) + ';"color1","' + cor + '";"empty","zero"})';
}
function ranking_(sh, s, titulo, mapa, cor, vazio) {
  moldeSlot_(sh, s, titulo, '');
  var arr = Object.keys(mapa).map(function (k) { return mapa[k]; }).sort(function (a, b) { return b.qtd - a.qtd; }).slice(0, 8);
  if (!arr.length) { sh.getRange(s.r + 2, s.c, 1, 8).merge().setValue(vazio).setFontColor(COR.cinzaClaro).setHorizontalAlignment('center'); return; }
  var max = arr[0].qtd;
  arr.forEach(function (it, i) {
    var row = s.r + 1 + i;
    sh.getRange(row, s.c, 1, 5).merge().setValue((i + 1) + '.   ' + it.desc).setFontWeight('normal');
    sh.getRange(row, s.c + 6).setFormula(barra_(it.qtd, max, cor));
    sh.getRange(row, s.c + 7).setValue(it.qtd).setFontWeight('bold').setHorizontalAlignment('right');
    linhaSep_(sh, s, row);
  });
}
function quadroMetas_(sh, s, equipes, regsPeriodo, f) {
  moldeSlot_(sh, s, 'Metas do mês por equipe', f.mes ? MESES[f.mes - 1] + ' ' + f.ano : 'ano ' + f.ano);
  var porEq = {}; regsPeriodo.forEach(function (r) { porEq[r.equipe] = (porEq[r.equipe] || 0) + r.pontos; });
  equipes.filter(function (t) { return t.ativa; }).slice(0, 8).forEach(function (t, i) {
    var row = s.r + 1 + i, p = porEq[t.nome] || 0, meta = t.meta || 0, pct = meta ? p / meta : 0;
    var cor = pct >= 1 ? COR.verdeOk : (pct >= 0.8 ? COR.ambar : COR.verde);
    sh.getRange(row, s.c, 1, 2).merge().setValue(t.nome).setFontWeight('bold');
    sh.getRange(row, s.c + 3, 1, 3).merge().setFormula(barra_(Math.min(p, meta || p), meta || Math.max(p, 1), cor));
    sh.getRange(row, s.c + 6).setValue(fmtNum_(p)).setHorizontalAlignment('right').setFontColor(COR.cinza);
    sh.getRange(row, s.c + 7).setValue(meta && f.mes ? Math.round(pct * 100) + '%' : '—').setFontWeight('bold').setHorizontalAlignment('right');
    linhaSep_(sh, s, row);
  });
}

  moldeSlot_(sh, s, 'Pendências para retorno', pend.length + ' em aberto');

  [[0, 1, 'OS'], [1, 1, 'EQUIPE'], [3, 3, 'OCORRÊNCIA · REFERÊNCIA'], [6, 1, 'LOCAL'], [7, 1, 'STATUS']].forEach(function (x) {

  });
  linhaSep_(sh, s, h);
  if (!pend.length) { sh.getRange(h + 2, s.c, 1, 8).merge().setValue('Nenhuma ocorrência parcial ou pendente no período.').setFontColor(COR.cinzaClaro).setHorizontalAlignment('center'); return; }

    sh.getRange(row, s.c + 3, 1, 3).merge().setValue(p.tipo + (p.local ? ' · ' + p.local : '')).setFontSize(9).setWrap(false);

    else sh.getRange(row, s.c + 6).setValue('—').setFontColor(COR.cinzaClaro);

  });
}

function atualizarRelatorio() {
  var sh = aba_(CFG.ABA_RELATORIO); if (!sh) return;
  var eq = String(sh.getRange('K5').getValue()), mes = MESES.indexOf(String(sh.getRange('L5').getValue())) + 1, ano = Number(sh.getRange('M5').getValue());
  if (!mes) mes = new Date().getMonth() + 1; if (!ano) ano = new Date().getFullYear();
  var regsTodos = lerRegistros_();
  var regs = regsTodos.filter(function (r) { return r.equipe === eq && r.data.getFullYear() === ano && r.data.getMonth() + 1 === mes; }).sort(function (a, b) { return a.data - b.data; });
  var equipe = lerEquipes_().filter(function (t) { return t.nome === eq; })[0] || { meta: 0 };
  var pts = regs.reduce(function (s, r) { return s + r.pontos; }, 0), meta = equipe.meta || 0, pct = meta ? pts / meta : 0;
  var dias = {}; regs.forEach(function (r) { dias[chaveDia_(r.data)] = true; });
  // status
  var est = !meta ? ['Sem meta definida', COR.cinzaClaro] : pct >= 1 ? ['Meta atingida', COR.verdeOk] : pct >= 0.8 ? ['Perto da meta', COR.ambar] : ['Abaixo da meta', COR.laranja];
  sh.getRange('K7').setValue('●  ' + est[0]).setFontColor(est[1]);
  var big = (meta ? Math.round(pct * 100) + '%' : fmtNum_(pts)), resto = meta ? '   ' + fmtNum_(pts) + ' de ' + fmtNum_(meta) + ' pontos · faltam ' + fmtNum_(Math.max(0, meta - pts)) : '   pontos no mês';
  sh.getRange('H8').setRichTextValue(SpreadsheetApp.newRichTextValue().setText(big + resto)
    .setTextStyle(0, big.length, ts_({ size: 24, bold: true })).setTextStyle(big.length, (big + resto).length, ts_({ size: 10, color: COR.cinza })).build());
  sh.getRange('H10').setFormula(barra_(Math.min(pts, meta || pts), meta || Math.max(pts, 1), est[1] === COR.cinzaClaro ? COR.verde : est[1]));

  sh.getRange('L11').setValue(regs.length);
  sh.getRange('M11').setValue(Object.keys(dias).length);
  // comissão por técnico
  var com = calcComissoes_(ano, mes, regsTodos).filter(function (p) { return p.equipeBase === eq || p.equipesReforco.indexOf(eq) !== -1; });
  var teto = com.length ? com[0].teto : lerRegrasComissao_().teto;
  sh.getRange('L13').setValue('teto ' + fmtR_(teto) + ' · faixas em Equipes & Metas');
  var linhasC = sh.getRange('H15:N18'); linhasC.clearContent(); linhasC.setBorder(false, false, false, false, false, false);
  com.slice(0, 4).forEach(function (p, i) {
    var row = 15 + i;

    sh.getRange(row, 10).setValue(p.dias).setHorizontalAlignment('right');
    sh.getRange(row, 11).setValue(fmtNum_(p.pontos)).setHorizontalAlignment('right');
    sh.getRange(row, 12).setValue(p.faixa + (p.obs ? ' · ' + p.obs : '') + (p.faltam ? ' · faltam ' + p.faltam : '')).setFontSize(9).setFontColor(p.obs ? '#8a3a0a' : COR.cinza);
    sh.getRange(row, 13).setValue(fmtR_(p.valor)).setFontWeight('bold').setHorizontalAlignment('right');
    sh.getRange(row, 8, 1, 7).setBorder(null, null, true, null, null, null, COR.linha, SpreadsheetApp.BorderStyle.SOLID);
  });
  if (!com.length) sh.getRange('H15').setValue('Sem técnicos cadastrados nesta equipe.').setFontColor(COR.cinzaClaro).setFontWeight('normal');
  sh.getRange('H19').setValue(com.length ? 'Pontos de cada técnico = pontos da equipe nos dias em que trabalhou × 2 ÷ pessoas no dia.' : '').setFontSize(8).setFontColor(COR.cinzaClaro);

  var ini = REL.dados, lr = sh.getMaxRows();

  area.breakApart(); area.clear(); area.clearDataValidations(); area.setBackground(COR.fundo).setFontFamily(FONTE).setFontSize(10);

function abrirDetalhesLinha_(linhaReg, equipe) {
  var reg = lerRegistros_().filter(function (r) { return r.linha === linhaReg; })[0];

    transporte: reg.transporte, obs: reg.obs, pontos: fmtNum_(reg.pontos), servicos: servs, ocorrencias: ocorrenciasDoRegistro_(reg), alerta: reg.alerta };

function abrirRelatorioMensal() {
  var sh = aba_(CFG.ABA_RELATORIO);
  var eq = String(sh.getRange('K5').getValue()), mes = MESES.indexOf(String(sh.getRange('L5').getValue())) + 1, ano = Number(sh.getRange('M5').getValue());
  if (!eq || !mes || !ano) { SpreadsheetApp.getUi().alert('Selecione a equipe, o mês e o ano no Relatório.'); return; }
  var cad = lerCadastros_(), conf = lerConfig_(), regsTodos = lerRegistros_();
  var regs = regsTodos.filter(function (r) { return r.equipe === eq && r.data.getFullYear() === ano && r.data.getMonth() + 1 === mes; });
  var serv = {}, oco = {}, tr = {}, total = 0;
  function add(m, nome, qtd, un, pts) { var k = norm_(nome); if (!m[k]) m[k] = { desc: nome, qtd: 0, un: un || '', pontos: 0 }; m[k].qtd += qtd; m[k].pontos += pts; }
  regs.forEach(function (r) {
    total += r.pontos;
    String(r.servicos || '').split('|').forEach(function (s) { s = s.trim(); if (!s) return; var it = partesServico_(s); add(serv, it.nome, it.qtd, it.un, calcularPontos_(s, '', '', cad, conf).pontos); });
    ocorrenciasDoRegistro_(r).forEach(function (o) { add(oco, o.tipo, 1, '', calcularPontos_('', o.tipo + ':', '', cad, conf).pontos); });
    var t = String(r.transporte || '');
    if (t && t !== 'Não' && t !== 'Nenhum') { var mt = t.match(/^(.*)\s\(\s*(\d+)\s*viage/i); add(tr, mt ? mt[1].trim() : t, mt ? +mt[2] : 1, 'viagem', calcularPontos_('', '', t, cad, conf).pontos); }
  });
  function arr(m) { return Object.keys(m).map(function (k) { var x = m[k]; return { desc: x.desc, qtd: fmtNum_(x.qtd) + (x.un ? ' ' + x.un : ''), pontos: fmtNum_(x.pontos) }; }).sort(function (a, b) { return a.desc.localeCompare(b.desc); }); }
  var com = calcComissoes_(ano, mes, regsTodos).filter(function (p) { return p.equipeBase === eq || p.equipesReforco.indexOf(eq) !== -1; })
    .map(function (p) { return { nome: p.nome, dias: p.dias, pontos: fmtNum_(p.pontos), faixa: p.faixa, valor: fmtR_(p.valor), obs: p.obs }; });
  var dados = { equipe: eq, periodo: MESES[mes - 1] + ' de ' + ano, os: regs.length, total: fmtNum_(total), servicos: arr(serv), ocorrencias: arr(oco), transportes: arr(tr), comissao: com };
  var t = HtmlService.createTemplateFromFile('RelatorioMensalModal');
  t.dadosJson = JSON.stringify(dados).replace(/</g, '\\u003c');
  SpreadsheetApp.getUi().showModalDialog(t.evaluate().setWidth(1000).setHeight(700), 'Relatório do mês · ' + eq);
}

function preencherUsadaEm_() {
  var med = nr_('CONF_MEDIDAS'), aj = nr_('CONF_AJUSTES'); if (!med) return;
  var cad = lerCadastros_(), conf = lerConfig_();
  var todos = cad.servicos.concat(cad.ocorrencias).filter(function (x) { return x.ativoFlag; });
  var out = med.getValues().map(function (r) {
    var ch = String(r[0]).split('·').map(norm_);
    var usados = todos.filter(function (s) { var m = norm_(s.medida); return ch.indexOf(m) !== -1 || (ch.indexOf('metro') !== -1 && (m === 'm' || m.indexOf('metro') === 0)); }).map(function (s) { return s.nome; });
    return [usados.length ? usados.join(', ') : '(nenhum ativo)'];
  });
  med.offset(0, 2, out.length, 1).setValues(out);
  if (aj) {
    var mServ = {}; cad.servicos.forEach(function (s) { mServ[norm_(s.nome)] = s; });
    var reg = aj.getValues().map(function (r) {
      if (!r[0]) return [''];
      var s = mServ[norm_(r[0])]; if (!s) return ['serviço não encontrado'];
      var m = norm_(s.medida), rg = regraMedida_(conf, s.medida), pct = num_(r[5]) === null ? 100 : num_(r[5]);
      if (m === 'm' || m.indexOf('metro') === 0) { var mpp = (rg && rg.mpp) || 40; return [(rg ? rg.regra : 'Por metro') + ' · 1 ponto a cada ' + fmtNum_(mpp * 100 / Math.max(pct, 1)) + ' m']; }
      return [(rg ? rg.regra : 'Por quantidade informada') + ' · ' + (num_(s.pontos) === null ? 'sem pontos' : fmtNum_(num_(s.pontos) * pct / 100) + ' pts')];
    });
    aj.offset(0, 4, reg.length, 1).setValues(reg);
  }
}

function instalarInfraControl() {
  var ss = ss_(), props = PropertiesService.getDocumentProperties();
  if (!props.getProperty('backupV3')) {
    var copia = ss.copy('Cópia de segurança — ' + ss.getName() + ' — ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd-MM-yyyy HH:mm'));
    props.setProperty('backupV3', copia.getUrl());
  }
  var antigos = lerEquipesAntigas_(aba_(CFG.ABA_EQUIPES));
  montarConfig_(antigos.pin);
  montarEquipes_(antigos);
  estilizarCadastros_();
  preencherUsadaEm_();
  estilizarRegistros_();
  recalcular_(null, null);
  montarRelatorio_();
  montarPainel_();
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'aoEditar') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('aoEditar').forSpreadsheet(ss).onEdit().create();
  aplicarCabecalhos_();
  [CFG.ABA_PAINEL, CFG.ABA_CAD, CFG.ABA_EQUIPES, CFG.ABA_RELATORIO, CFG.ABA_REGISTROS, CFG.ABA_CONFIG].forEach(function (n, i) {
    var s = aba_(n); if (s) { ss.setActiveSheet(s); ss.moveActiveSheet(i + 1); }
  });
  aba_(CFG.ABA_PAINEL).activate();
  return 'ok · cópia de segurança: ' + props.getProperty('backupV3');
}
